## Copycat migration test, phase 1 (run on 0.7.0, before Copycats+): place Create: Connected copycats
## with a material in a force-loaded chunk. Usage: test-server.py ... --commands scripts/copycat-migration-1-place.txt
forceload add 0 48 15 63
#sleep 5
setblock 0 100 48 create_connected:copycat_slab{Item:{id:"minecraft:oak_planks",count:1},Material:{Name:"minecraft:oak_planks"}}
setblock 2 100 48 create_connected:copycat_beam{Item:{id:"minecraft:stone_bricks",count:1},Material:{Name:"minecraft:stone_bricks"}}
setblock 4 100 48 create_connected:copycat_vertical_step{Item:{id:"minecraft:andesite",count:1},Material:{Name:"minecraft:andesite"}}
setblock 6 100 48 create_connected:copycat_board{Item:{id:"minecraft:spruce_planks",count:1},Material:{Name:"minecraft:spruce_planks"}}
setblock 8 100 48 create_connected:copycat_block{Item:{id:"minecraft:bricks",count:1},Material:{Name:"minecraft:bricks"}}
setblock 10 100 48 create_connected:copycat_stairs{Item:{id:"minecraft:deepslate_tiles",count:1},Material:{Name:"minecraft:deepslate_tiles"}}
setblock 12 100 48 create_connected:copycat_wall{Item:{id:"minecraft:cobblestone",count:1},Material:{Name:"minecraft:cobblestone"}}
execute if block 0 100 48 create_connected:copycat_slab run say VT_CC_COPYCAT_SLAB_PLACED
data get block 0 100 48
execute if block 2 100 48 create_connected:copycat_beam run say VT_CC_COPYCAT_BEAM_PLACED
data get block 2 100 48
execute if block 4 100 48 create_connected:copycat_vertical_step run say VT_CC_COPYCAT_VERTICAL_STEP_PLACED
data get block 4 100 48
execute if block 6 100 48 create_connected:copycat_board run say VT_CC_COPYCAT_BOARD_PLACED
data get block 6 100 48
execute if block 8 100 48 create_connected:copycat_block run say VT_CC_COPYCAT_BLOCK_PLACED
data get block 8 100 48
execute if block 10 100 48 create_connected:copycat_stairs run say VT_CC_COPYCAT_STAIRS_PLACED
data get block 10 100 48
execute if block 12 100 48 create_connected:copycat_wall run say VT_CC_COPYCAT_WALL_PLACED
data get block 12 100 48
save-all flush
stop
